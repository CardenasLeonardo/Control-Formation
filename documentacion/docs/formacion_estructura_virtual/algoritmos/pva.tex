{\small
\begin{algorithm}[H]
\SetKwInput{KwData}{Datos}
\SetKwInput{KwResult}{Resultado}
\SetAlgorithmName{Algoritmo}{algoritmo}{Lista de algoritmos}
\caption{PVA usando LiDAR}
\KwData{LiDAR, $u_{ref}=(v_{ref},\omega_{ref})$}
\KwResult{$u^*=(v,\omega)$}
$\mathcal{C} \leftarrow \{|v| \le v_{max},\; |\omega| \le \omega_{max}\}$\;
\For{cada rayo $i$}{
    $A_i \leftarrow \cos(\phi_i)$\;
    $B_i \leftarrow r_p\sin(\phi_i)$\;
    \If{$0 < d_i \le d_{inf}$}{
        $C_i \leftarrow \xi \dfrac{d_i-d_{safe}}{d_{inf}-d_{safe}}$\;
        $\mathcal{C} \leftarrow \mathcal{C} \cup \{A_i v + B_i \omega \le C_i\}$\;
    }
}
$u^* = \arg\min_{u} \dfrac{1}{2}\|u-u_{ref}\|^2$\;
\KwRet $u^*$
\end{algorithm}
